{{- /* Resizes a post's cover image for listings. Args: dict "page" "width" "height" Returns dict {src, srcset, width, height, alt} or false when the post has no cover. */ -}} {{- $p := .page -}} {{- $out := false -}} {{- with $p.Params.cover }}{{ with $p.Resources.Get . -}} {{- $w := $.width }}{{ $h := $.height -}} {{- $one := .Fill (printf "%dx%d Center webp" $w $h) -}} {{- $two := .Fill (printf "%dx%d Center webp" (mul $w 2) (mul $h 2)) -}} {{- $out = dict "src" $one.RelPermalink "srcset" (printf "%s %dw, %s %dw" $one.RelPermalink $one.Width $two.RelPermalink $two.Width) "width" $one.Width "height" $one.Height "alt" ($p.Params.coverAlt | default "") -}} {{- else -}} {{- errorf "%s: cover image %q not found next to the post's index.md" $p.File.Path . -}} {{- end }}{{ end -}} {{- return $out -}}
