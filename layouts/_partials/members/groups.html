{{- /* Reads the members data file (data/members.yaml, or the file named by site.Params.membersData) and returns a list of tier groups: [ { tier: , members: [, …] }, … ] Only tiers that have at least one member are returned. A tier with sortByName: true lists its members alphabetically. On the live site (production), members marked `pending: true` are left out; Netlify previews and `hugo server` show them, so they can be reviewed. Arg: "home" → only tiers with showOnHome: true; anything else → all tiers. */ -}} {{- $scope := . -}} {{- $data := index hugo.Data (site.Params.membersData | default "members") | default dict -}} {{- $members := $data.members | default slice -}} {{- if hugo.IsProduction }}{{ $members = where $members "pending" "ne" true }}{{ end -}} {{- $groups := slice -}} {{- $tierIDs := slice -}} {{- range $t := $data.tiers }} {{- $tierIDs = $tierIDs | append $t.id -}} {{- if or (ne $scope "home") $t.showOnHome }} {{- with where $members "tier" $t.id }} {{- $list := . }}{{ if $t.sortByName }}{{ $list = sort $list "name" }}{{ end }} {{- $groups = $groups | append (dict "tier" $t "members" $list) }} {{- end -}} {{- end }} {{- end }} {{- if eq $scope "home" }}{{ range $members }} {{- if not (in $tierIDs .tier) }}{{ errorf "data/members.yaml: %q has tier %q, which isn't one of the tier ids (%s). Tier ids are case-sensitive." .name .tier (delimit $tierIDs ", ") }}{{ end -}} {{- end }}{{ end }} {{- return $groups -}}
