{{- /* Returns the social-sharing image for a page as a dict {url,width,height,alt}. Blog posts use their cover image, cropped to 1200×630 (the size LinkedIn and others display). Everything else uses site.Params.ogImage. */ -}} {{- $alt := .Params.coverAlt | default .Title | default site.Title -}} {{- $img := false -}} {{- $anchor := "Center" -}} {{- if and .IsPage (eq .Section "blog") }}{{ $img = partial "blog/cover-resource.html" . }}{{ $anchor = partial "blog/cover-anchor.html" . }}{{ end -}} {{- if not $img }}{{ $img = resources.Get site.Params.ogImage }}{{ $alt = site.Title }}{{ end -}} {{- $out := dict -}} {{- with $img -}} {{- if reflect.IsImageResourceProcessable . -}} {{- $c := .Fill (printf "1200x630 %s jpg q85" $anchor) -}} {{- $out = dict "url" $c.Permalink "width" $c.Width "height" $c.Height "alt" $alt -}} {{- end -}} {{- end -}} {{- return $out -}}
